        button:hover { background-color: #2980b9; }

    <select id="alkaneSelect">

        <option value="4">Бутан</option>

        <option value="7">Гептан</option>
        <option value="8">Октан</option>
        <option value="benzene">Бензол</option>
    </select>
    <button id="resetBtn">Сбросить реакцию</button>
    <div class="info">
        <div id="nameDisplay" style="font-weight: 600; color: #1a1a1a;">Этан</div>
        <div id="formulaDisplay" class="formula-box">C₂H₆</div>
        <div id="errorDisplay" class="error-msg"></div>
    </div>

<canvas id="moleculeCanvas"></canvas>

<script>
    const canvas = document.getElementById('moleculeCanvas');
    const ctx = canvas.getContext('2d');
    const select = document.getElementById('alkaneSelect');
    const resetBtn = document.getElementById('resetBtn');
    const errorDisplay = document.getElementById('errorDisplay');

    let mode = 'linear'; // 'linear' или 'benzene'
    let n = 2;
    let bondStates = []; 
    let anims = []; 
    const spacing = 120;

    function getSub(num) { return num.toString().split('').map(d => "₀₁₂₃₄₅₆₇₈₉"[d] || d).join(''); }

    function updateUI() {
        if (mode === 'benzene') {
            document.getElementById('nameDisplay').innerText = "Бензол";
            document.getElementById('formulaDisplay').innerText = `C₆H₆`;
            return;
        }
        const roots = ["", "Мет", "Эт", "Проп", "Бут", "Пент", "Гекс", "Гепт", "Окт"];
        let hCount = 2 * n + 2;
        let doubles = [], triples = [];
        bondStates.forEach((s, i) => {
            if (s === 2) { doubles.push(i + 1); hCount -= 2; }
            if (s === 3) { triples.push(i + 1); hCount -= 4; }
        });

        let name = roots[n];
        let resName = (doubles.length === 0 && triples.length === 0) ? name + "ан" : 
                      (name + (doubles.length > 1 || triples.length > 1 ? "а" : "") + 
                      (doubles.length > 0 ? (doubles.length > 1 ? "диен" : "ен") + "-" + doubles.join(',') : "") +
                      (triples.length > 0 ? (triples.length > 1 ? "диин" : "ин") + "-" + triples.join(',') : ""));
        
        document.getElementById('nameDisplay').innerText = resName;
        document.getElementById('formulaDisplay').innerText = `C${n > 1 ? getSub(n) : ''}H${getSub(hCount)}`;
    }

    function init() {
        const val = select.value;
        errorDisplay.innerText = "";
        if (val === 'benzene') {
            mode = 'benzene';
            updateUI();
            draw();
        } else {
            mode = 'linear';
            n = parseInt(val);
            bondStates = new Array(n - 1).fill(1);
            anims = new Array(n - 1).fill(null);
            updateUI();
            draw();
        }
    }

    function drawAtom(x, y, symbol, color, isH = false) {
        ctx.beginPath();
        ctx.arc(x, y, isH ? 18 : 24, 0, Math.PI * 2);
        ctx.fillStyle = color; ctx.fill();
        ctx.strokeStyle = "#333"; ctx.lineWidth = 2; ctx.stroke();
        ctx.fillStyle = "#000"; ctx.font = `bold ${isH ? 14 : 16}px Arial`;
        ctx.textAlign = "center"; ctx.textBaseline = "middle";
        ctx.fillText(symbol, x, y);
    }

    function drawBondCC(x1, y1, x2, y2, level) {
        ctx.strokeStyle = "#555"; ctx.lineWidth = 3;
        if (level === 1) { ctx.beginPath(); ctx.moveTo(x1, y1); ctx.lineTo(x2, y2); ctx.stroke(); }
        else if (level === 2) {
            let dx = y2 - y1, dy = x1 - x2;
            let len = Math.sqrt(dx*dx + dy*dy);
            dx /= len; dy /= len;
            ctx.beginPath(); ctx.moveTo(x1 + dx*5, y1 + dy*5); ctx.lineTo(x2 + dx*5, y2 + dy*5); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(x1 - dx*5, y1 - dy*5); ctx.lineTo(x2 - dx*5, y2 - dy*5); ctx.stroke();
        } else if (level === 3) {
            ctx.beginPath(); ctx.moveTo(x1, y1); ctx.lineTo(x2, y2); ctx.stroke();
            let dx = y2 - y1, dy = x1 - x2;
            let len = Math.sqrt(dx*dx + dy*dy);
            dx /= len; dy /= len;
            ctx.beginPath(); ctx.moveTo(x1 + dx*9, y1 + dy*9); ctx.lineTo(x2 + dx*9, y2 + dy*9); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(x1 - dx*9, y1 - dy*9); ctx.lineTo(x2 - dx*9, y2 - dy*9); ctx.stroke();
        }
    }

    function drawLinear() {
        const startX = n === 1 ? canvas.width / 2 : (canvas.width - (n - 1) * spacing) / 2;
        const startY = canvas.height / 2;
        for (let i = 0; i < n - 1; i++) drawBondCC(startX + i * spacing, startY, startX + (i + 1) * spacing, startY, bondStates[i]);
        for (let i = 0; i < n; i++) {
            const cx = startX + i * spacing, cy = startY;
            if (n === 1 || i === 0) { ctx.beginPath(); ctx.moveTo(cx, cy); ctx.lineTo(cx - spacing, cy); ctx.stroke(); drawAtom(cx - spacing, cy, 'H', '#fff', true); }
            if (n === 1 || i === n - 1) { ctx.beginPath(); ctx.moveTo(cx, cy); ctx.lineTo(cx + spacing, cy); ctx.stroke(); drawAtom(cx + spacing, cy, 'H', '#fff', true); }
            let hU = true, hD = true, uO = 0, dO = 0, dU = true, dD = true;
            if (i < n - 1 && bondStates[i] >= 2) { if (anims[i] && anims[i].s === 1) { uO = -anims[i].p * 6; dU = false; } else hU = false; }
            if (i > 0 && bondStates[i-1] === 3) { if (anims[i-1] && anims[i-1].s === 2) { uO = -anims[i-1].p * 6; dU = false; } else hU = false; }
            if (i > 0 && bondStates[i-1] >= 2) { if (anims[i-1] && anims[i-1].s === 1) { dO = anims[i-1].p * 6; dD = false; } else hD = false; }
            if (i < n - 1 && bondStates[i] === 3) { if (anims[i] && anims[i].s === 2) { dO = anims[i].p * 6; dD = false; } else hD = false; }
            if (hU) { if (dU) { ctx.beginPath(); ctx.moveTo(cx, cy); ctx.lineTo(cx, cy - spacing); ctx.stroke(); } drawAtom(cx, cy - spacing + uO, 'H', '#fff', true); }
            if (hD) { if (dD) { ctx.beginPath(); ctx.moveTo(cx, cy); ctx.lineTo(cx, cy + spacing); ctx.stroke(); } drawAtom(cx, cy + spacing + dO, 'H', '#fff', true); }
            drawAtom(cx, cy, 'C', '#bdc3c7');
        }
    }

    function drawBenzene() {
        const cx = canvas.width / 2, cy = canvas.height / 2, r = 100, hR = 170;
        const pts = [];
        for (let i = 0; i < 6; i++) {
            pts.push({ x: cx + r * Math.cos(i * Math.PI / 3), y: cy + r * Math.sin(i * Math.PI / 3) });
        }
        for (let i = 0; i < 6; i++) {
            drawBondCC(pts[i].x, pts[i].y, pts[(i + 1) % 6].x, pts[(i + 1) % 6].y, (i % 2 === 0 ? 2 : 1));
            const hx = cx + hR * Math.cos(i * Math.PI / 3), hy = cy + hR * Math.sin(i * Math.PI / 3);
            ctx.beginPath(); ctx.moveTo(pts[i].x, pts[i].y); ctx.lineTo(hx, hy); ctx.stroke();
            drawAtom(hx, hy, 'H', '#fff', true);
        }
        pts.forEach(p => drawAtom(p.x, p.y, 'C', '#bdc3c7'));
    }

    function draw() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        if (mode === 'benzene') drawBenzene(); else drawLinear();
        let active = false;
        anims.forEach((a, idx) => { if (a) { a.p += 4; active = true; if (a.p >= 100) anims[idx] = null; } });
        if (active) requestAnimationFrame(draw);
    }

    canvas.addEventListener('mousedown', (e) => {
        if (mode === 'benzene') return;
        const startX = (canvas.width - (n - 1) * spacing) / 2;
        const startY = canvas.height / 2;
        for (let i = 0; i < n - 1; i++) {
            const xMid = startX + i * spacing + spacing / 2;
            if (Math.abs(e.clientX - xMid) < 40 && Math.abs(e.clientY - startY) < 30) {
                let next = bondStates[i] + 1;
                if (next > 3) return;
                if (((i > 0 ? bondStates[i-1] : 1) + next <= 4) && ((i < n - 2 ? bondStates[i+1] : 1) + next <= 4)) {
                    bondStates[i] = next; anims[i] = { s: next - 1, p: 1 }; updateUI(); draw();
                } else {
                    errorDisplay.innerText = "Ошибка: Валентность превысит 4!";
                    setTimeout(() => errorDisplay.innerText = "", 2000);
                }
            }
        }
    });

    resetBtn.addEventListener('click', init);
    select.onchange = init;
    window.onresize = () => { canvas.width = window.innerWidth; canvas.height = window.innerHeight; draw(); };
    window.onresize(); init();
</script>
</body>
